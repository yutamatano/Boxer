cmake_minimum_required(VERSION 3.16)
project(standalone_neon_slam LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(OpenCV 4.4 REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Pangolin REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
# Match Eigen's allocation ABI in the library, runner, and both dependencies.
add_compile_definitions(EIGEN_MAX_ALIGN_BYTES=16 EIGEN_MAX_STATIC_ALIGN_BYTES=16)
add_subdirectory("${ORB_SLAM3_ROOT}/Thirdparty/DBoW2" dbow2)
add_subdirectory("${ORB_SLAM3_ROOT}/Thirdparty/g2o" g2o)
target_link_libraries(DBoW2 Boost::serialization)
file(GLOB ORB_SOURCES "${ORB_SLAM3_ROOT}/src/*.cc" "${ORB_SLAM3_ROOT}/src/*.cpp" "${ORB_SLAM3_ROOT}/src/CameraModels/*.cpp")
option(NEON_ORPHAN_GUARD "Do not cull a keyframe without a spanning-tree parent" OFF)
if(NEON_ORPHAN_GUARD)
  # The pristine checkout remains untouched. This optional one-condition guard
  # prevents SetBadFlag -> ChangeParent(nullptr) -> AddChild's null dereference.
  set(KEYFRAME_SOURCE "${ORB_SLAM3_ROOT}/src/KeyFrame.cc")
  file(READ "${KEYFRAME_SOURCE}" KEYFRAME_TEXT)
  string(REPLACE "if(mnId==mpMap->GetInitKFid())" "if(!mpParent && mnId!=mpMap->GetInitKFid())\n            std::cerr << \"Standalone orphan cull skipped: \" << mnId << std::endl;\n        if(mnId==mpMap->GetInitKFid() || !mpParent)" KEYFRAME_GUARDED "${KEYFRAME_TEXT}")
  if(KEYFRAME_TEXT STREQUAL KEYFRAME_GUARDED)
    message(FATAL_ERROR "Pinned KeyFrame guard location was not found")
  endif()
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/KeyFrame_guard.cc" "${KEYFRAME_GUARDED}")
  list(REMOVE_ITEM ORB_SOURCES "${KEYFRAME_SOURCE}")
  list(APPEND ORB_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/KeyFrame_guard.cc")
  # Inertial culling rewires temporal links BEFORE calling SetBadFlag. Reject
  # the orphan at the caller too, before modifying those links/preintegrations.
  set(MAPPING_SOURCE "${ORB_SLAM3_ROOT}/src/LocalMapping.cc")
  file(READ "${MAPPING_SOURCE}" MAPPING_TEXT)
  string(REPLACE "const vector<MapPoint*> vpMapPoints = pKF->GetMapPointMatches();" "if(!pKF->GetParent()) {\n            std::cerr << \"Standalone orphan cull skipped before IMU rewiring: \" << pKF->mnId << std::endl;\n            continue;\n        }\n        const vector<MapPoint*> vpMapPoints = pKF->GetMapPointMatches();" MAPPING_GUARDED "${MAPPING_TEXT}")
  if(MAPPING_TEXT STREQUAL MAPPING_GUARDED)
    message(FATAL_ERROR "Pinned culling guard location was not found")
  endif()
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/LocalMapping_guard.cc" "${MAPPING_GUARDED}")
  list(REMOVE_ITEM ORB_SOURCES "${MAPPING_SOURCE}")
  list(APPEND ORB_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/LocalMapping_guard.cc")
endif()
add_library(standalone_orb SHARED ${ORB_SOURCES})
target_compile_definitions(standalone_orb PUBLIC COMPILEDWITHC11)
target_include_directories(standalone_orb PUBLIC "${ORB_SLAM3_ROOT}" "${ORB_SLAM3_ROOT}/include" "${ORB_SLAM3_ROOT}/include/CameraModels" "${ORB_SLAM3_ROOT}/Thirdparty/Sophus" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(standalone_orb PUBLIC DBoW2 g2o Eigen3::Eigen ${OpenCV_LIBS} pango_display Boost::serialization OpenSSL::Crypto Threads::Threads)
add_executable(neon_standalone neon_runner.cc)
set_property(TARGET neon_standalone PROPERTY CXX_STANDARD 17)
target_link_libraries(neon_standalone PRIVATE standalone_orb)
if(NEON_ORPHAN_GUARD)
  add_executable(test_orphan_guard test_orphan_guard.cc)
  target_link_libraries(test_orphan_guard PRIVATE standalone_orb)
  enable_testing()
  add_test(NAME orphan_keyframe_guard COMMAND test_orphan_guard)
endif()
